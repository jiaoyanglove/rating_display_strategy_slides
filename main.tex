\begin{frame}{研究背景}
\textbf{在线购物中，消费者如何判断产品质量？}

\vspace{8pt}
\begin{itemize}
    \item 网购时，消费者\pos{无法亲身体验}产品
    \item \HL{在线评论}成为了解产品质量的重要途径
    \item 据统计，超过90\%的消费者在购买前会查看评论
\end{itemize}

\vspace{12pt}
\textbf{平台面临的设计问题}

\vspace{6pt}
当平台销售\HL{多个相似产品}时（如同品牌不同型号），评论应该如何展示？

\section{文献综述}
%==============================================================================

\begin{frame}{信息设计与贝叶斯说服}
\small
\textbf{Kamenica \& Gentzkow (2011) — 贝叶斯说服}
\begin{itemize}
    \item \textbf{内容}：发送者承诺于特定信号结构，通过"凹化"价值函数设计最优信息揭示
    \item \textbf{区别}：其信号结构外生设计；本文中评分由消费者购买\HL{内生产生}
    \item \textbf{借鉴}：贝叶斯更新的基本框架
\end{itemize}

\vspace{6pt}
\textbf{Bergemann \& Morris (2019) — 信息设计统一框架}
\begin{itemize}
    \item \textbf{内容}：将贝叶斯说服推广为一般信息设计问题
    \item \textbf{区别}：标准凹化方法难以直接适用于内生评论生成机制
    \item \textbf{借鉴}：信息结构影响均衡结果的分析视角
\end{itemize}

\vspace{6pt}
\textbf{Lizzeri (1999) — 中介信息披露}
\begin{itemize}
    \item \textbf{内容}：垄断认证中介的最优策略通常是减少信息揭示
    \item \textbf{区别}：不涉及多产品定价；本文关注粗糙化如何影响\pos{差别定价能力}
\end{itemize}
\end{frame}

\begin{frame}{平台评论系统设计}
\small
\textbf{Xiao et al. (2022) — 评论集群展示}
\begin{itemize}
    \item \textbf{内容}：研究评论按产品属性分组展示的最优策略
    \item \textbf{区别}：关注\con{属性维度}的信息组织；本文关注\pos{产品版本}之间的信息混合
    \item \textbf{借鉴}：辅助函数 $g(\mu, \sigma)$ 的技术处理方法
\end{itemize}

\vspace{8pt}
\textbf{Shi et al. (2023) — 平台声誉系统设计}
\begin{itemize}
    \item \textbf{内容}：研究平台声誉系统的最优披露粒度
    \item \textbf{发现}：部分披露能同时激励卖家提升质量和增加交易量
    \item \textbf{区别}：模型中只有\con{同质产品}，没有涉及垂直差异化下的跨产品信息混合
\end{itemize}

\vspace{8pt}
\textbf{Dai et al. (2018) — Yelp 评分聚合}
\begin{itemize}
    \item \textbf{内容}：实证发现不同评分聚合方式带来效率差异
    \item \textbf{借鉴}：现实背景支持；但尚未形式化刻画分开/聚合展示制度
\end{itemize}
\end{frame}

\begin{frame}{多产品定价与本文贡献}
\small
\textbf{Mussa \& Rosen (1978) — 垂直差异化定价}
\begin{itemize}
    \item \textbf{内容}：垄断者通过质量与价格菜单实施二级价格歧视
    \item \textbf{区别}：假设消费者完全知道质量；本文中消费者需要\HL{通过评分学习质量}
\end{itemize}

\vspace{8pt}
\textbf{Yu et al. (2016) — 评论驱动的动态定价}
\begin{itemize}
    \item \textbf{内容}：评论可能导致企业在初期控制销量（"信息生成的负外部性"）
    \item \textbf{借鉴}：聚合展示下增加一种产品销量会\con{稀释另一种产品的信号质量}
\end{itemize}

\vspace{10pt}
\begin{block}{本文贡献}
\begin{enumerate}
    \item 首次在\HL{两期垂直差异化}框架内研究评论展示策略选择
    \item 将信息的\pos{内生生成过程}纳入信息设计框架
    \item 刻画展示制度、消费者学习、平台定价之间的联系
\end{enumerate}
\end{block}
\end{frame}

\subsection{模型设定}

\begin{frame}{博弈时间线}
\textbf{参与者}：垄断平台（销售产品B和产品G）、消费者

\begin{center}

\vspace{6pt}
\textbf{分析框架}：给定展示策略 $\psi$，求解平台最优定价决策
\end{frame}

\begin{frame}{研究框架：四个核心问题}
\begin{enumerate}
    \item \textbf{消费者如何决策？}
    \begin{itemize}
        \item 第一期：基于先验信念 $q_p$ 决定是否购买
        \item 第二期：观察评分信号 $r_\psi$，贝叶斯更新后验信念 $v_{\psi\delta}$，再决定购买
    \end{itemize}

    \vspace{6pt}
    \item \textbf{平台如何决策？}
    \begin{itemize}
        \item $t=0$：选择展示策略 $\psi \in \{S, I\}$，制定第一期价格 $p_{\delta 1}$
        \item $t=1$：观察评分后，制定第二期最优价格 $p^*_{\delta 2} = v_{\psi\delta}/(2h)$
    \end{itemize}

    \vspace{6pt}
    \item \textbf{如何比较两种策略？}
    \begin{itemize}
        \item 对称情形：解析比较方差 $\sigma_S^2$ vs $\sigma_I^2$、利润 $\Pi_S^*$ vs $\Pi_I^*$
        \item 一般情形：数值模拟验证参数敏感性
    \end{itemize}

    \vspace{6pt}
    \item \textbf{其他考量因素}
    \begin{itemize}
        \item 消费者福利：$CS_S^*$ vs $CS_I^*$
        \item 社会总福利：$W_S^*$ vs $W_I^*$
        \item 扩展：评论精度非对称性 $\zeta_B \neq \zeta_G$
    \end{itemize}
\end{enumerate}
\end{frame}

\begin{frame}{产品质量与消费者感知}
\textbf{产品质量}

两产品质量独立同分布（对称设定）：
\[
q_B \sim N(q_p, 1/\tau), \quad q_G \sim N(q_p, 1/\tau)
\]
其中 $q_p$ 为先验均值，$\tau$ 为先验精度

\vspace{10pt}
\textbf{消费者感知}

消费者 $i$ 对产品 $\delta$ 的感知质量带有噪声：
\[
q_{\delta,i} \sim N(q_\delta, 1/\zeta)
\]
其中 $\zeta$ 为感知精度

\vspace{10pt}
\textbf{需求函数}
\[
k_\delta = m(v_\delta - hp_\delta), \quad \delta \in \{B, G\}
\]
$v_\delta$：感知价值，$p_\delta$：价格，$h$：价格敏感度，$m$：市场规模
\end{frame}

\begin{frame}{评论展示策略}
平台选择展示策略 $\psi \in \{S, I\}$，购买者中比例 $\theta$ 发布评论

\vspace{8pt}
\begin{columns}[T]
\column{0.48\textwidth}
\begin{alertblock}{分开展示 (Separate)}
各产品评分独立计算：
\[
r_\delta = \frac{1}{\theta k_\delta}\sum_{i=1}^{\theta k_\delta} q_{\delta,i}
\]
条件分布：
\[
r_\delta|q_\delta \sim N\left(q_\delta, \frac{1}{k_\delta\theta\zeta}\right)
\]
\end{alertblock}

\column{0.48\textwidth}
\begin{exampleblock}{聚合展示 (Integrate)}
按购买量加权平均：
\[
r_I = \frac{k_B r_B + k_G r_G}{k_B + k_G}
\]
条件分布：
\[
r_I|(q_B,q_G) \sim N\left(\bar{q}_w, \frac{1}{(k_B+k_G)\theta\zeta}\right)
\]
其中 $\bar{q}_w = \frac{k_B q_B + k_G q_G}{k_B + k_G}$
\end{exampleblock}
\end{columns}
\end{frame}

\subsection{消费者学习：贝叶斯分析}

\begin{frame}{分开展示：贝叶斯更新}
\textbf{推导过程}

先验与似然的PDF：
\[
f(q_B) \propto \exp\left(-\frac{\tau}{2}(q_B - q_p)^2\right), \quad
f(r_B|q_B) \propto \exp\left(-\frac{k_B\theta\zeta}{2}(r_B - q_B)^2\right)
\]

由贝叶斯公式 $f(q_B|r_B) \propto f(r_B|q_B)f(q_B)$，得\pos{后验分布}：
\[
q_B|r_B \sim N\left(\frac{\tau q_p + k_B\theta\zeta \cdot r_B}{\tau + k_B\theta\zeta}, \frac{1}{\tau + k_B\theta\zeta}\right)
\]

\vspace{6pt}
\textbf{后验均值}（第二期消费者感知价值）：
\[
v_{SB} = \frac{\tau q_p + k_B\theta\zeta \cdot r_B}{\tau + k_B\theta\zeta}, \quad
v_{SG} = \frac{\tau q_p + k_G\theta\zeta \cdot r_G}{\tau + k_G\theta\zeta}
\]

\vspace{4pt}
\HL{关键特征}：$v_{SB}$ 与 $v_{SG}$ \textbf{相互独立}
\end{frame}

\begin{frame}{聚合展示：贝叶斯更新}
\textbf{推导过程}

消费者仅观察聚合评分 $r_I$，需对另一产品质量积分（边缘化）：
\[
f(r_I|q_B) = \int_{-\infty}^{+\infty} f(r_I|q_B, q_G)f(q_G)dq_G
\]

应用\HL{高斯积分公式} $\int e^{Aq^2 + Bq + C}dq \propto \exp\left(-\frac{B^2}{4A} + C\right)$

\vspace{6pt}
\textbf{后验分布}：
\[
q_B|r_I \sim N\left(\frac{(k_B+k_G)k_B\theta\zeta \cdot r_I - k_Bk_G\theta\zeta \cdot q_p + (k_G^2\theta\zeta + (k_B+k_G)\tau)q_p}{(k_B^2+k_G^2)\theta\zeta + (k_B+k_G)\tau}, \sigma_{post}^2\right)
\]

\vspace{6pt}
\con{关键特征}：$v_{IB}$ 与 $v_{IG}$ \textbf{相互耦合}——两产品信息混合在同一信号中
\end{frame}

\begin{frame}{后验方差的先验分析}
利用\textbf{全方差公式}：$\text{Var}[v_{\psi\delta}] = \text{Var}[q_\delta] - \mathbb{E}[\text{Var}[q_\delta|r_\psi]]$

\vspace{8pt}
\begin{columns}[T]
\column{0.48\textwidth}
\begin{alertblock}{分开展示}
\[
\mathbb{E}[v_{S\delta}] = q_p
\]
\[
\sigma_{S\delta}^2 = \frac{k_\delta\theta\zeta}{\tau^2 + k_\delta\tau\theta\zeta}
\]
\end{alertblock}

\column{0.48\textwidth}
\begin{exampleblock}{聚合展示}
\[
\mathbb{E}[v_{I\delta}] = q_p
\]
\[
\sigma_{I\delta}^2 = \frac{k_\delta^2\theta\zeta}{(k_B^2+k_G^2)\tau\theta\zeta + (k_B+k_G)\tau^2}
\]
\end{exampleblock}
\end{columns}

\vspace{10pt}
\begin{block}{命题：信息稀释效应}
对于任意 $k_B, k_G > 0$，有 $\sigma_{I\delta}^2 < \sigma_{S\delta}^2$。

聚合展示引入\con{评论来源不确定性}，降低了消费者学习效率。
\end{block}
\end{frame}

\subsection{平台决策：利润优化}

\begin{frame}{利润结构}
\textbf{第一期利润}

由需求函数反解价格：$p_{\delta 1} = \frac{m q_p - k_\delta}{hm}$，利润为：
\[
\Pi_{\psi 1} = \frac{(m q_p - k_B)k_B}{hm} + \frac{(m q_p - k_G)k_G}{hm}
\]

\vspace{8pt}
\textbf{第二期利润}

给定后验感知价值 $v_{\psi\delta}$，平台最大化 $\Pi_{\psi\delta 2} = p \cdot m(v_{\psi\delta} - hp)$

一阶条件得最优价格与利润：
\[
p^*_{\psi\delta 2} = \max\left\{0, \frac{v_{\psi\delta}}{2h}\right\}, \quad
\Pi^*_{\psi 2} = \frac{m \cdot \max\{0, v_{\psi B}\}^2}{4h} + \frac{m \cdot \max\{0, v_{\psi G}\}^2}{4h}
\]

\vspace{8pt}
\textbf{总期望利润}：
\[
\Pi_\psi = \Pi_{\psi 1} + \int_0^{+\infty} \frac{m v_{\psi B}^2}{4h} f_{\psi B}(v)dv + \int_0^{+\infty} \frac{m v_{\psi G}^2}{4h} f_{\psi G}(v)dv
\]
\end{frame}

\begin{frame}{最优定价决策：一般情形}
引入辅助函数（Xiao 2022）：$g(\mu, \sigma) = \int_{-\mu/\sigma}^{+\infty} z(z + \frac{\mu}{\sigma})\phi(z)dz$，性质：$g$ 关于 $\sigma$ 递减

\vspace{6pt}
\textbf{一般情形的一阶条件}（隐函数方程）

\vspace{4pt}
\begin{alertblock}{分开展示：两个独立方程}
\vspace{-8pt}
\begin{align*}
k_{SB}^* &= \frac{m_1 q_L}{2} + \frac{m_1 m_2\theta\zeta}{8} g(q_L, \sigma_{SB}) \frac{\tau^2}{(\tau^2 + k_{SB}^*\tau\theta\zeta)^2}\\[2pt]
k_{SG}^* &= \frac{m_1 q_H}{2} + \frac{m_1 m_2\theta\zeta}{8} g(q_H, \sigma_{SG}) \frac{\tau^2}{(\tau^2 + k_{SG}^*\tau\theta\zeta)^2}
\end{align*}
\end{alertblock}

\vspace{-4pt}
\begin{exampleblock}{聚合展示：两个耦合方程}
\vspace{-8pt}
\begin{align*}
k_{IB}^* &= \frac{m_1 q_L}{2} + \frac{m_1 m_2\theta\zeta}{8}\left[g(q_L, \sigma_{IB})\cdot A_B + g(q_H, \sigma_{IG})\cdot B_B\right]\\[2pt]
k_{IG}^* &= \frac{m_1 q_H}{2} + \frac{m_1 m_2\theta\zeta}{8}\left[g(q_L, \sigma_{IB})\cdot A_G + g(q_H, \sigma_{IG})\cdot B_G\right]
\end{align*}
\vspace{-8pt}
其中 $A_\delta, B_\delta$ 为 $(k_{IB}, k_{IG}, \tau, \theta, \zeta)$ 的复杂函数
\end{exampleblock}

\vspace{4pt}
\con{问题}：一般情形下为隐式方程组，\textbf{无法解析求解} $\Rightarrow$ 考虑对称特殊情形
\end{frame}

\subsection{策略比较}

\begin{frame}{对称情形：方差的 2:1 比例}
考察对称情形：$q_L = q_H = q$，$m_1, m_2$ 相同。此时最优解也对称：$k_{SB}^* = k_{SG}^* = k_S^*$，$k_{IB}^* = k_{IG}^* = k_I^*$

\vspace{6pt}
\begin{block}{引理：对称情形下的方差公式}
\begin{align*}
\sigma_S^2 &= \frac{k_S\theta\zeta}{\tau^2 + k_S\tau\theta\zeta}, &
\sigma_I^2 &= \frac{k_I\theta\zeta}{2(\tau^2 + k_I\tau\theta\zeta)}
\end{align*}
特别地，当 $k_S = k_I = k$ 时，$\sigma_I^2 = \sigma_S^2 / 2$
\end{block}

\vspace{6pt}
\textbf{经济含义}：
\begin{itemize}
    \item 聚合展示将样本量从 $k$ 增加到 $2k$
    \item 但消费者\con{无法区分评论来源}
    \item 来源不确定性使有效信息量\con{减半}
    \item 该 2:1 比例与质量差距 $\Delta q$ \textbf{无关}
\end{itemize}
\end{frame}

\begin{frame}{对称情形：一阶条件与利润排序}
\begin{block}{命题：对称情形下的一阶条件}
\begin{align*}
k_S^* &= \frac{m_1 q}{2} + \frac{m_1 m_2\theta\zeta\tau^2}{8}\cdot\frac{g(q, \sigma_S)}{(\tau^2 + k_S^*\tau\theta\zeta)^2}\\[4pt]
k_I^* &= \frac{m_1 q}{2} + \frac{m_1 m_2\theta\zeta\tau^2}{8}\cdot\frac{g(q, \sigma_I)}{\pos{2}(\tau^2 + k_I^*\tau\theta\zeta)^2}
\end{align*}
聚合展示的信息激励项分母多了因子 \pos{2}
\end{block}

\vspace{6pt}
\begin{block}{命题：对称情形下的排序}
\[
k_S^* > k_I^*, \quad \Pi_S^* > \Pi_I^*, \quad CS_S^* > CS_I^*, \quad W_S^* > W_I^*
\]
分开展示在购买量、利润、消费者剩余、社会福利四个维度上\textbf{均严格占优}
\end{block}
\end{frame}

\begin{frame}{分开展示普遍占优：经济解释}
\begin{block}{命题：分开展示的普遍优势}
在本文模型设定下，对于\textbf{所有测试的参数配置}：
\[
\Pi_S^* > \Pi_I^*, \quad CS_S^* > CS_I^*, \quad W_S^* > W_I^*
\]
\end{block}

\vspace{6pt}
\textbf{为什么聚合展示无法逆转劣势？}
\begin{enumerate}
    \item \pos{信息价值为正}：精确信号 $\Rightarrow$ 状态依存定价 $\Rightarrow$ 期望收益增加
    \item \con{样本规模优势不足}：来源不确定性成本 $>$ 样本合并收益
    \item \con{交叉效应有限}：最优购买量差异仅 5--7\%，不足以改变排序
\end{enumerate}

\vspace{6pt}
\textbf{核心洞察}：聚合展示的劣势来源于\HL{评论来源不确定性}本身，而非产品质量差异。即使 $q_L = q_H$，分开展示仍严格占优。
\end{frame}

\subsection{福利分析}

\begin{frame}{福利分析：利润与福利目标一致}
\textbf{消费者剩余}
\[
CS_{\psi 2} = \frac{m_{B2} \cdot \E[\max\{0, v_{\psi B}^2\}]}{8h} + \frac{m_{G2} \cdot \E[\max\{0, v_{\psi G}^2\}]}{8h}
\]

\vspace{6pt}
\begin{block}{命题 5.6：分开展示的福利优势}
在本文模型设定下，分开展示策略\textbf{同时}最大化平台利润、消费者剩余和社会总福利：
\[
\Pi_S^* > \Pi_I^*, \quad CS_S^* > CS_I^*, \quad W_S^* > W_I^*
\]
\end{block}

\vspace{6pt}
\textbf{重要结论}：
\begin{itemize}
    \item 平台利润最大化与社会福利最大化\HL{指向同一策略}
    \item \pos{不存在利润-福利冲突}
    \item 与信息设计文献中常见的权衡不同
\end{itemize}
\end{frame}

\subsection{模型扩展}

\begin{frame}{扩展：评论精度非对称性}
\textbf{扩展动机}
\begin{itemize}
    \item 高质量产品用户评论可能更详细、更精确
    \item 不同产品的用户群体评论质量可能不同
\end{itemize}

\vspace{6pt}
\textbf{扩展设定}：$\zeta_B \neq \zeta_G$，定义精度比 $\rho = \zeta_G / \zeta_B$

\vspace{6pt}
\textbf{直观分析}
\begin{itemize}
    \item 分开展示：各产品分别利用各自精度 $\zeta_B$、$\zeta_G$
    \item 聚合展示：使用\con{加权调和平均}精度 $\bar{\zeta} = \frac{k_B + k_G}{k_B/\zeta_B + k_G/\zeta_G}$
    \item 调和平均对低精度敏感，低精度评论拖累整体信息质量
\end{itemize}

\vspace{6pt}
\con{问题}：非对称精度下一阶条件更复杂，\textbf{无解析解} $\Rightarrow$ 数值模拟
\end{frame}

\begin{frame}{扩展：数值分析结果}
\begin{center}
\small
\begin{tabular}{cccc}
\toprule

\midrule

0.50 & 2.5634 & 2.4412 & 0.1222 \\
1.00 & 2.6355 & 2.5741 & 0.0613 \\
2.00 & 2.7089 & 2.5623 & 0.1466 \\

\bottomrule
\end{tabular}
\end{center}

\vspace{8pt}
\begin{block}{数值结论：精度非对称性的影响}
\begin{itemize}
    \item 分开展示在所有测试配置 $\rho \in [0.2, 5.0]$ 下均产生更高利润
    \item 精度非对称性（$\rho \neq 1$）\HL{放大}分开展示的利润优势
    \item 基准结论对评论精度非对称性具有稳健性
\end{itemize}
\end{block}
\end{frame}

\section{结论与展望}
%==============================================================================

\begin{frame}{主要结论}
\begin{enumerate}
    \item \textbf{聚合展示降低信息精度}
    \begin{itemize}
        \item 引入"评论来源不确定性"，消费者学习效率降低
    \end{itemize}

    \vspace{6pt}
    \item \textbf{分开展示普遍占优}
    \begin{itemize}
        \item 对称情形下信息效率 2:1
        \item 所有测试参数下均占优
    \end{itemize}

    \vspace{6pt}
    \item \textbf{利润与福利目标一致}
    \begin{itemize}
        \item 分开展示同时最大化利润、消费者剩余和社会福利
        \item 不存在利润-福利冲突
    \end{itemize}

    \vspace{6pt}
    \item \textbf{评论精度非对称性强化结论}
    \begin{itemize}
        \item 精度差异越大，分开展示优势越明显
    \end{itemize}
\end{enumerate}
\end{frame}

\begin{frame}{管理启示与未来方向}
\textbf{管理启示}
\begin{itemize}
    \item 垂直差异化产品组合：优先采用\pos{分开展示}
    \item 高质量评论是信息资产，避免与低质量评论混合稀释
    \item 可通过评论标签、评论者等级提升评论精度差异化
\end{itemize}

\vspace{10pt}
\textbf{未来研究方向}
\begin{itemize}
    \item \textbf{模型拓展}：多产品（$>2$）、动态评论积累、竞争平台
    \item \textbf{机制拓展}：消费者异质性学习、评论操纵成本
    \item \textbf{实证研究}：利用平台数据检验理论预测
\end{itemize}
\end{frame}

\begin{frame}{贝叶斯更新推导细节}
\textbf{分开展示下的后验}

给定 $n = k_\delta \theta$ 个评论信号，后验分布为：
\[
q_\delta \mid \{r_i\} \sim N\left(\frac{\tau \bar{q}_\delta + n\zeta\bar{r}}{\tau + n\zeta}, \frac{1}{\tau + n\zeta}\right)
\]

\vspace{6pt}
\textbf{后验均值的方差推导}

由于 $\bar{r} = q_\delta + \bar{\epsilon}$，后验均值 $\hat{q}_\delta$ 的方差为：
\[
\Var(\hat{q}_\delta) = \left(\frac{n\zeta}{\tau + n\zeta}\right)^2 \cdot \frac{1}{n\zeta} = \frac{n\zeta}{(\tau + n\zeta)^2}
\]

代入 $n = k_\delta \theta$ 并整理：
\[
\sigma_{S\delta}^2 = \frac{k_\delta\theta\zeta}{\tau^2 + k_\delta\tau\theta\zeta}
\]
\end{frame}

\begin{frame}{聚合展示可能占优的扩展方向}
\textbf{本文基准模型下分开展示普遍占优，但以下扩展可能改变结论：}

\vspace{6pt}
\begin{enumerate}
    \item \textbf{消费者异质性学习能力}
    \begin{itemize}
        \item "朴素型"消费者偏好评论数量 $\Rightarrow$ 聚合展示可能占优
    \end{itemize}

    \vspace{4pt}
    \item \textbf{评论操纵与信誉成本}
    \begin{itemize}
        \item 分开展示使负面信号更可见 $\Rightarrow$ 信誉成本高时聚合可能占优
    \end{itemize}

    \vspace{4pt}
    \item \textbf{多期动态与产品多样性}
    \begin{itemize}
        \item 聚合展示可保护弱势产品 $\Rightarrow$ 长期多样性价值高时可能占优
    \end{itemize}

    \vspace{4pt}
    \item \textbf{竞争性平台}
    \begin{itemize}
        \item 展示策略可作为差异化工具 $\Rightarrow$ 可能存在非对称均衡
    \end{itemize}
\end{enumerate}
